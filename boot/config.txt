# ==============================================================================
# Omarchy Quatro Pi 5 - Hardware Boot Configuration
# Target: Raspberry Pi 5 (Broadcom BCM2712 Cortex-A76)
# ==============================================================================

# Enable 64-bit ARM execution mode
arm_64bit=1

# Early serial / UART debugging and 2nd stage bootloader logging
enable_uart=1
uart_2ndstage=1

# Hardware peripheral auto-detection
camera_auto_detect=1
display_auto_detect=1

# Graphics & Audio
dtoverlay=vc4-kms-v3d
dtparam=audio=on

# Argon ONE / NEO 5 case fan & power MCU I2C communication
dtparam=i2c_arm=on

# PCIe Gen 3 NVMe SSD Support (~850-900 MB/s for 512GB NVMe)
dtparam=pciex1
dtparam=pciex1_gen=3

# ------------------------------------------------------------------------------
# Pi 5 Specific Kernel Configuration
# ------------------------------------------------------------------------------
[pi5]
# 16KB page size kernel for Raspberry Pi 5
kernel=kernel8.img
auto_initramfs=1

# ------------------------------------------------------------------------------
# Universal Fallthrough
# ------------------------------------------------------------------------------
[all]
